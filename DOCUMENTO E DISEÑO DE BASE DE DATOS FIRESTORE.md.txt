Principios:
1. Denormalización controlada para lecturas rápidas
2. Subcolecciones para datos 1:N acotados
3. Colecciones raíz para datos que crecen indefinidamente
4. Composite indexes para queries frecuentes
5. Counter fields para aggregaciones (evitar count queries)

// firestore.rules
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper functions
    function isAuthenticated() {
      return request.auth != null;
    }

    function isOwner(userId) {
      return isAuthenticated() && request.auth.uid == userId;
    }

    function hasRole(role) {
      return isAuthenticated() &&
             request.auth.token.role == role;
    }

    function isAdmin() {
      return hasRole('ADMIN');
    }

    function isMentor() {
      return hasRole('MENTORA');
    }

    function isEmprendedora() {
      return hasRole('EMPRENDEDORA');
    }

    // ============ USERS ============
    match /users/{userId} {
      allow read: if isAuthenticated();
      allow create: if isOwner(userId);
      allow update: if isOwner(userId) || isAdmin();
      allow delete: if isAdmin();

      match /devices/{deviceId} {
        allow read: if isOwner(userId);
        allow write: if isOwner(userId);
      }

      match /notifications/{notifId} {
        allow read: if isOwner(userId);
        allow update: if isOwner(userId) && request.resource.data.diff(resource.data).affectedKeys()
                       .hasOnly(['read']);
        allow create: if false; // only via Cloud Functions
        allow delete: if false;
      }

      match /feedback/{feedbackId} {
        allow read: if isOwner(userId) || isAdmin();
        allow create: if isOwner(userId);
      }
    }

    // ============ SCHOLARSHIP APPLICATIONS ============
    match /scholarship_applications/{appId} {
      allow read: if isAuthenticated() &&
                     (resource.data.userId == request.auth.uid || isAdmin());
      allow create: if isAuthenticated() && isEmprendedora()
                     && request.resource.data.userId == request.auth.uid
                     && request.resource.data.status == 'PENDIENTE';
      allow update: if isAdmin()
                     || (isOwner(resource.data.userId)
                         && request.resource.data.diff(resource.data).affectedKeys()
                            .hasOnly(['updatedAt']));
      allow delete: if false; // never delete, only change status
    }

    // ============ STORES ============
    match /stores/{storeId} {
      allow read: if true; // public stores
      allow create: if isAuthenticated();
      allow update: if isOwner(resource.data.ownerId) || isAdmin();
      allow delete: if isAdmin();

      match /products/{productId} {
        allow read: if true; // public products
        allow create: if isOwner(get(/databases/$(database)/documents/stores/$(storeId)).data.ownerId);
        allow update: if isOwner(get(/databases/$(database)/documents/stores/$(storeId)).data.ownerId)
                       || isAdmin();
        allow delete: if isOwner(get(/databases/$(database)/documents/stores/$(storeId)).data.ownerId)
                       || isAdmin();
      }

      match /orders/{orderId} {
        allow read: if isAuthenticated() &&
                     (resource.data.buyerId == request.auth.uid
                      || isOwner(get(/databases/$(database)/documents/stores/$(storeId)).data.ownerId)
                      || isAdmin());
        allow create: if isAuthenticated();
        allow update: if isOwner(get(/databases/$(database)/documents/stores/$(storeId)).data.ownerId)
                       || isAdmin();
      }

      match /pos_sessions/{sessionId} {
        allow read, write: if isOwner(get(/databases/$(database)/documents/stores/$(storeId)).data.ownerId)
                           || isAdmin();
      }
    }

    // ============ MENTORSHIP SESSIONS ============
    match /mentorship_sessions/{sessionId} {
      allow read: if isAuthenticated() &&
                     (resource.data.menteeId == request.auth.uid
                      || resource.data.mentorId == request.auth.uid
                      || isAdmin());
      allow create: if isAuthenticated() &&
                     (request.resource.data.menteeId == request.auth.uid
                      || request.resource.data.mentorId == request.auth.uid);
      allow update: if isAuthenticated() &&
                     (resource.data.menteeId == request.auth.uid
                      || resource.data.mentorId == request.auth.uid
                      || isAdmin());
      allow delete: if isAdmin();
    }

    // ============ MENTORS ============
    match /mentors/{mentorId} {
      allow read: if isAuthenticated();
      allow create: if isMentor() || isAdmin();
      allow update: if isOwner(resource.data.userId) || isAdmin();
      allow delete: if isAdmin();

      match /schedule_overrides/{overrideId} {
        allow read: if isAuthenticated();
        allow write: if isOwner(get(/databases/$(database)/documents/mentors/$(mentorId)).data.userId)
                    || isAdmin();
      }
    }

    // ============ MARKETPLACE LISTINGS ============
    match /marketplace_listings/{listingId} {
      allow read: if resource.data.status == 'APPROVED' || isAdmin()
                  || (isAuthenticated() && resource.data.ownerId == request.auth.uid);
      allow create: if isAuthenticated() &&
                     request.resource.data.ownerId == request.auth.uid;
      allow update: if isOwner(resource.data.ownerId) || isAdmin();
      allow delete: if isOwner(resource.data.ownerId) || isAdmin();
    }

    // ============ EVENTS ============
    match /events/{eventId} {
      allow read: if true; // public events
      allow create: if isAdmin();
      allow update: if isAdmin();
      allow delete: if isAdmin();

      match /registrations/{regId} {
        allow read: if isAuthenticated() &&
                     (resource.data.userId == request.auth.uid || isAdmin());
        allow create: if isAuthenticated() &&
                     request.resource.data.userId == request.auth.uid;
        allow update: if isAdmin();
        allow delete: if isOwner(resource.data.userId) || isAdmin();
      }
    }

    // ============ COMMUNITY POSTS ============
    match /community_posts/{postId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated();
      allow update: if isOwner(resource.data.authorId) || isAdmin();
      allow delete: if isOwner(resource.data.authorId) || isAdmin();

      match /comments/{commentId} {
        allow read: if isAuthenticated();
        allow create: if isAuthenticated();
        allow update: if isOwner(resource.data.authorId) || isAdmin();
        allow delete: if isOwner(resource.data.authorId) || isAdmin();
      }
    }

    // ============ CONVERSATIONS ============
    match /conversations/{conversationId} {
      allow read: if isAuthenticated() &&
                     request.auth.uid in resource.data.participants;
      allow create: if isAuthenticated() &&
                     request.auth.uid in request.resource.data.participants;
      allow update: if isAuthenticated() &&
                     request.auth.uid in resource.data.participants;

      match /messages/{messageId} {
        allow read: if isAuthenticated() &&
                     request.auth.uid in get(/databases/$(database)/documents/conversations/$(conversationId)).data.participants;
        allow create: if isAuthenticated() &&
                     request.auth.uid in get(/databases/$(database)/documents/conversations/$(conversationId)).data.participants
                     && request.resource.data.senderId == request.auth.uid;
      }
    }

    // ============ ANALYTICS ============
    match /analytics/{date} {
      allow read: if isAdmin();
      allow write: if false; // only via Cloud Functions
    }
  }
}
