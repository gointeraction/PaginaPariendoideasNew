// src/middleware.ts
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

const protectedPaths = ['/dashboard', '/kit-mujer-ceo', '/impulso', '/marketplace/publicar'];
const adminPaths = ['/admin'];

export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  
  // Verificar token en cookies
  const token = request.cookies.get('auth-token')?.value;
  
  if (protectedPaths.some(path => pathname.startsWith(path))) {
    if (!token) {
      return NextResponse.redirect(new URL('/login', request.url));
    }
  }
  
  if (adminPaths.some(path => pathname.startsWith(path))) {
    if (!token) {
      return NextResponse.redirect(new URL('/login', request.url));
    }
    // Verificar custom claims de admin
    // (decodificar token JWT y verificar role)
  }
  
  return NextResponse.next();
}

export const config = {
  matcher: ['/dashboard/:path*', '/kit-mujer-ceo/:path*', '/impulso/:path*', '/admin/:path*', '/marketplace/publicar/:path*'],
};
